cmake_minimum_required(VERSION 3.25)
project(sihopt VERSION 0.5.1 LANGUAGES CXX)
option(SIHOPT_WARNINGS_AS_ERRORS "Treat warnings as errors" ON)
option(SIHOPT_ENABLE_ASAN_UBSAN "Enable address/undefined sanitizers" OFF)
option(SIHOPT_ENABLE_TSAN "Enable thread sanitizer" OFF)
option(SIHOPT_BUILD_FUZZER "Build libFuzzer MPS target with Clang" OFF)
if(SIHOPT_ENABLE_ASAN_UBSAN AND SIHOPT_ENABLE_TSAN)
  message(FATAL_ERROR "ASan/UBSan and TSan are mutually exclusive")
endif()
add_library(sihopt_core STATIC src/foundation/build_info.cpp src/model/model.cpp src/io/mps.cpp src/verify/primal_verifier.cpp src/linalg/dense_lu.cpp src/linalg/sparse_basis.cpp src/transform/canonicalize.cpp src/lp/reference/revised_simplex.cpp src/verify/reference_lp_verifier.cpp src/lp/dual/dual_simplex.cpp)
target_include_directories(sihopt_core PUBLIC include)
target_compile_features(sihopt_core PUBLIC cxx_std_20)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(sihopt_core PRIVATE -Wall -Wextra -Wpedantic)
  if(SIHOPT_WARNINGS_AS_ERRORS)
    target_compile_options(sihopt_core PRIVATE -Werror)
  endif()
  if(SIHOPT_ENABLE_ASAN_UBSAN)
    target_compile_options(sihopt_core PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(sihopt_core PUBLIC -fsanitize=address,undefined)
  elseif(SIHOPT_ENABLE_TSAN)
    target_compile_options(sihopt_core PRIVATE -fsanitize=thread -fno-omit-frame-pointer)
    target_link_options(sihopt_core PUBLIC -fsanitize=thread)
  endif()
endif()
add_executable(sihopt-info apps/sihopt_info.cpp)
target_link_libraries(sihopt-info PRIVATE sihopt_core)
add_executable(sihopt-mps-inspect apps/sihopt_mps_inspect.cpp)
target_link_libraries(sihopt-mps-inspect PRIVATE sihopt_core)
enable_testing()
add_executable(foundation_test tests/foundation_test.cpp)
target_link_libraries(foundation_test PRIVATE sihopt_core)
add_executable(m1_test tests/m1_test.cpp)
target_link_libraries(m1_test PRIVATE sihopt_core)
add_executable(m1_edge_test tests/m1_edge_test.cpp)
target_link_libraries(m1_edge_test PRIVATE sihopt_core)
add_executable(mps_fuzz_smoke tests/fuzz/mps_fuzz_smoke.cpp)
target_link_libraries(mps_fuzz_smoke PRIVATE sihopt_core)
add_executable(m1_property_test tests/m1_property_test.cpp)
target_link_libraries(m1_property_test PRIVATE sihopt_core)
add_executable(m2_test tests/m2_test.cpp)
target_link_libraries(m2_test PRIVATE sihopt_core)
add_executable(m3_test tests/m3_test.cpp)
target_link_libraries(m3_test PRIVATE sihopt_core)
add_executable(m3_property_test tests/m3_property_test.cpp)
target_link_libraries(m3_property_test PRIVATE sihopt_core)
add_executable(m4_test tests/m4_test.cpp)
target_link_libraries(m4_test PRIVATE sihopt_core)
add_executable(m4_property_test tests/m4_property_test.cpp)
target_link_libraries(m4_property_test PRIVATE sihopt_core)
add_executable(m5_test tests/m5_test.cpp)
target_link_libraries(m5_test PRIVATE sihopt_core)
add_executable(m5_property_test tests/m5_property_test.cpp)
target_link_libraries(m5_property_test PRIVATE sihopt_core)
add_executable(m5_audit_regression_test tests/m5_audit_regression_test.cpp)
target_link_libraries(m5_audit_regression_test PRIVATE sihopt_core)
add_test(NAME foundation_contract COMMAND foundation_test)
add_test(NAME m1_model_parser_verifier COMMAND m1_test)
add_test(NAME m1_edge_cases COMMAND m1_edge_test)
add_test(NAME m1_fuzz_smoke COMMAND mps_fuzz_smoke)
add_test(NAME m1_randomized_properties COMMAND m1_property_test)
add_test(NAME m2_dense_lu_canonicalization COMMAND m2_test)
add_test(NAME m3_reference_primal_simplex COMMAND m3_test)
add_test(NAME m3_randomized_differential COMMAND m3_property_test)
add_test(NAME m4_dual_simplex COMMAND m4_test)
add_test(NAME m4_randomized_warm_starts COMMAND m4_property_test)
add_test(NAME m5_sparse_basis COMMAND m5_test)
add_test(NAME m5_sparse_update_properties COMMAND m5_property_test)
add_test(NAME m5_audit_regressions COMMAND m5_audit_regression_test)
add_test(NAME json_records COMMAND ${CMAKE_COMMAND} -E env python3 ${CMAKE_SOURCE_DIR}/scripts/validate_records.py ${CMAKE_SOURCE_DIR})
add_test(NAME no_future_solver_guard COMMAND ${CMAKE_COMMAND} -E env python3 ${CMAKE_SOURCE_DIR}/scripts/no_solver_guard.py ${CMAKE_SOURCE_DIR})
set_tests_properties(foundation_contract m1_model_parser_verifier m1_edge_cases m1_fuzz_smoke m1_randomized_properties m2_dense_lu_canonicalization m3_reference_primal_simplex m3_randomized_differential m4_dual_simplex m4_randomized_warm_starts m5_sparse_basis m5_sparse_update_properties json_records no_future_solver_guard PROPERTIES TIMEOUT 30)
if(SIHOPT_BUILD_FUZZER)
  if(NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang")
    message(FATAL_ERROR "libFuzzer target requires Clang")
  endif()
  add_executable(mps_fuzz tests/fuzz/mps_fuzz.cpp)
  target_link_libraries(mps_fuzz PRIVATE sihopt_core)
  target_compile_options(mps_fuzz PRIVATE -fsanitize=fuzzer,address,undefined)
  target_link_options(mps_fuzz PRIVATE -fsanitize=fuzzer,address,undefined)
  add_executable(sparse_basis_fuzz tests/fuzz/sparse_basis_fuzz.cpp)
  target_link_libraries(sparse_basis_fuzz PRIVATE sihopt_core)
  target_compile_options(sparse_basis_fuzz PRIVATE -fsanitize=fuzzer,address,undefined)
  target_link_options(sparse_basis_fuzz PRIVATE -fsanitize=fuzzer,address,undefined)
endif()
